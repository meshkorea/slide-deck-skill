export async function charge(order: Order) {
  const total = sum(order.items)          // 합계는 기존 로직 그대로
  if (total <= 0) throw new Error("빈 주문")

  const res = await gateway.pay(order.id, total)

  return { ok: res.approved, at: res.paidAt }
}
